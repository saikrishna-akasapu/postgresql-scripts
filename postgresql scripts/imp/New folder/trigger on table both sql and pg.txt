SELECT T.NAME AS NAME,TR.NAME AS TRNAME,TR.*
FROM SYS.TRIGGERS TR
JOIN SYS.TABLES T ON(TR.PARENT_ID=T.OBJECT_ID)
WHERE T.NAME IN('FO_BILL','FO_BILL_SRV');
