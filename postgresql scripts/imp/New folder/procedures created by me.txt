select n.nspname as schema_name,
p.proname as function_name ,
pg_get_functiondef(p.oid)as function_def
from pg_proc p
join pg_namespace n on (n.oid=p.pronamespace)
where  
  (n.nspname ='stpl_hims_p_m_dbo')
and pg_get_functiondef(p.oid) ilike '%vinod%'
and pg_get_functiondef(p.oid) ilike '%itemstock%'
and  p.proname  not like 'pr_%'
group by n.nspname,p.proname,p.oid
-- AND P.PRONAME ILIKE '%WARD_TRAN%'




select n.nspname as schema_name,
p.proname as function_name ,
pg_get_functiondef(p.oid)as function_def
from pg_proc p
join pg_namespace n on (n.oid=p.pronamespace)
where  
  (n.nspname ='stpl_hims_p_m_dbo')
and pg_get_functiondef(p.oid) ilike '%PHANI%'
and pg_get_functiondef(p.oid) ilike '%CALL%'
and  p.proname  not like 'pr_%'
and  p.proname    like 'fn_tr_%'
group by n.nspname,p.proname,p.oid
