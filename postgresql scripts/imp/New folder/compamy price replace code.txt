				IS_SERVICE_NAME_EDIT

				(
					SELECT
						SERVICE_NAME,
						SERVICE_ID,
						SERVICE_PRICE_ID,
						(
							CASE
								WHEN (
									(COALESCE(IS_GST_APPLICABLE, 'N')) = 'Y'
									AND ('YES') = 'YES'
									AND (COALESCE(IS_TAX_INCLUDE_EXCLUDE, 'N')) = 'Y'
								) THEN PRICE
								WHEN (
									(COALESCE(IS_GST_APPLICABLE, 'N')) = 'Y'
									AND ('YES') = 'YES'
									AND (COALESCE(IS_TAX_INCLUDE_EXCLUDE, 'N')) = 'N'
								) THEN CAST(
									((PRICE * (TAX_PCT / 100)) + PRICE) AS NUMERIC(18, 2)
								)
								ELSE PRICE
							END
						) AS PRICE_WITH_GST,
						(
							CASE
								WHEN (
									(COALESCE(IS_GST_APPLICABLE, 'N')) = 'Y'
									AND ('YES') = 'YES'
									AND (COALESCE(IS_TAX_INCLUDE_EXCLUDE, 'N')) = 'Y'
								) THEN CAST(
									(PRICE - (PRICE / ((TAX_PCT / 100) + 1))) AS NUMERIC(18, 2)
								)
								WHEN (
									(COALESCE(IS_GST_APPLICABLE, 'N')) = 'Y'
									AND ('YES') = 'YES'
									AND (COALESCE(IS_TAX_INCLUDE_EXCLUDE, 'N')) = 'N'
								) THEN CAST((PRICE * (TAX_PCT / 100)) AS NUMERIC(18, 2))
								ELSE 0
							END
						) AS TAX_AMOUNT,
						(
							CASE
								WHEN (
									(COALESCE(IS_GST_APPLICABLE, 'N')) = 'Y'
									AND ('YES') = 'YES'
									AND (COALESCE(IS_TAX_INCLUDE_EXCLUDE, 'N')) = 'Y'
								) THEN CAST((PRICE / ((TAX_PCT / 100) + 1)) AS NUMERIC(18, 2))
								WHEN (
									(COALESCE(IS_GST_APPLICABLE, 'N')) = 'Y'
									AND ('YES') = 'YES'
									AND (COALESCE(IS_TAX_INCLUDE_EXCLUDE, 'N')) = 'N'
								) THEN PRICE
								ELSE PRICE
							END
						) AS PRICE,
						ORG_PRICE,
						DOCTOR_PRICE,
						(
							CASE
								WHEN (
									(COALESCE(IS_GST_APPLICABLE, 'N')) = 'Y'
									AND ('YES') = 'YES'
									AND (COALESCE(IS_TAX_INCLUDE_EXCLUDE, 'N')) = 'Y'
								) THEN CAST(
									(EMERGENCY_PRICE / ((TAX_PCT / 100) + 1)) AS NUMERIC(18, 2)
								)
								WHEN (
									(COALESCE(IS_GST_APPLICABLE, 'N')) = 'Y'
									AND ('YES') = 'YES'
									AND (COALESCE(IS_TAX_INCLUDE_EXCLUDE, 'N')) = 'N'
								) THEN EMERGENCY_PRICE
								ELSE EMERGENCY_PRICE
							END
						) AS EMERGENCY_PRICE,
						(
							CASE
								WHEN (
									(COALESCE(IS_GST_APPLICABLE, 'N')) = 'Y'
									AND ('YES') = 'YES'
									AND (COALESCE(IS_TAX_INCLUDE_EXCLUDE, 'N')) = 'Y'
								) THEN EMERGENCY_PRICE
								WHEN (
									(COALESCE(IS_GST_APPLICABLE, 'N')) = 'Y'
									AND ('YES') = 'YES'
									AND (COALESCE(IS_TAX_INCLUDE_EXCLUDE, 'N')) = 'N'
								) THEN CAST(
									(
										(EMERGENCY_PRICE * (TAX_PCT / 100)) + EMERGENCY_PRICE
									) AS NUMERIC(18, 2)
								)
								ELSE PRICE
							END
						) AS EMERGENCY_PRICE_WITH_GST,
						SERVICE_TYPE_NAME,
						SERVICE_GROUP_ID,
						SERVICECLASS_ID,
						SERVICE_CD,
						SERVICE_GROUP,
						SERVICE_TYPE_ID,
						DOCTOR_ID,

						EFFECT_FROM_DT,

						IS_FOREIGN_SERVICE,
						IS_CLINICAL_HIST_REQ,
						GENDER_ID,
						HISTORY_TYPE_ID,

						QYT_EDIT,

						EMP_PERCENT,

						EMP_PRICE,
						ORG_PERCENT_PRICE,
						FROM_DAYS,
						TO_DAYS,
						IS_POST,

						IS_SHOW,

						UTILIZED_REFRL_QTY,
						MAX_OPT_SERVICES_ALLOWED,
						UTILZ_CMP_RULE_QTY,
						CMO_RUL_QTY,
						IS_REMARKS_MANDATORY,
						IS_SRV_GUIDELINES_REQUIRED,
						IS_CONSENT_FORM,
						IS_SRV_CHECKLIST_REQUIRED,
						IS_ADDITIONAL,
						DISCOUNT_PERCENT,
						IS_SRV_INSTRUCTION_REQ,
						DOC_HOL_STATUS,
						CMP_PKG_EXIST,
						SERVICE_UNICODE,

						STATUS,
						FROM_DT,
						TO_DT,
						SERVICE_QUESTION_TEMPLATE_ID,
						PRESCRIPTION_CHARGE_TYPE_ID,
						IS_DOCTOR_REQUIRED,
						IS_SCHEDULE_REQUIRED,
						VISIT_TYPE,
						BILLING_HEAD_ID,
						DOCTOR_PCT,
						ORG_PCT,
						IS_FAVORITE_SERVICE,
						IS_GST_APPLICABLE,
						(
							CASE
								WHEN (COALESCE('YES', 'NO')) = 'NO' THEN 0
								ELSE TAX_PCT
							END
						) AS TAX_PCT,
						SGST_PCT,
						CGST_PCT,
						SAC_CD,

						CNCSN_RULE_NAME,
						DEFINE_BY_ID,

						IS_SERVICE_NAME_EDIT
					FROM
						(
							SELECT DISTINCT
								SERVICE_NAME,
								S.SERVICE_ID,
								SP.SERVICE_PRICE_ID,
								COALESCE(CS.PRICE, SP.PRICE) PRICE,
								SP.ORG_PRICE,
								SP.DOCTOR_PRICE,
								(
									SELECT
										COALESCE(SR.PRICE, 0)
									FROM
										STPL_HIMS_P_M_DBO.SERVICE_PRICE SR
										LEFT JOIN STPL_HIMS_P_M_DBO.COMPANY CC ON 
										(
											SR.TARIFF_ID = SP.TARIFF_ID
											AND CC.COMPANY_ID = 47
											AND CC.LOC_ID = 1
											AND (CC.RECORD_STATUS) = 'A'
										)
									WHERE
										SR.SERVICE_ID = 6
										AND 
										(
											PATIENT_CLASS_ID = 2
											OR PATIENT_CLASS_ID = 3
											OR PATIENT_CLASS_ID = 0

										AND CONSULTATION_TYPE_ID = 2
										AND (SR.RECORD_STATUS) = 'A'
										AND SR.GRP_ID = 1
										AND SR.ORG_ID = 1
										AND SR.LOC_ID = 1
									LIMIT
										1

								S.SERVICE_TYPE_NAME,

									TO_CHAR(
										S.EFFECT_FROM_DT + (330 || 'MINUTE')::INTERVAL,
										'YYYY-MM-DD HH24:MI:SS'
									),
									1,
									64
								) EFFECT_FROM_DT,
								SUBSTRING(
									TO_CHAR(
										S.EFFECT_TO_DT + (330 || 'MINUTE')::INTERVAL,
										'YYYY-MM-DD HH24:MI:SS'
									),
									1,
									64
								) EFFECT_TO_DT,
								S.RECORD_STATUS,

								SP.TARIFF_ID,

												) - DATE_TRUNC
												(
													'DAY',
													(
														COALESCE
														(
															V.SRV_POSTED_DT,
															CURRENT_TIMESTAMP(3) AT TIME ZONE 'UTC'
														)
													)::TIMESTAMP
												)
										) >= COALESCE(NO_NEED_DAYS, '0') THEN 'N'
										ELSE 'Y'
									END
								) NO_NEED_SRV,
								CONCERN_FORM_REQ,
								SUBSTRING(
									TO_CHAR(
										V.SRV_POSTED_DT + (330 || 'MINUTE')::INTERVAL,
										'YYYY-MM-DD HH24:MI:SS'
									),
									1,
									64
								) PRIV_SRV_POSTED_DT,
								NO_NEED_DAYS,
								CASE
									WHEN 47 > 0 THEN SP.IS_QTY_EDIT_OP
									ELSE QYT_EDIT
								END QYT_EDIT,
								CASE
									WHEN 47 > 0 THEN SP.IS_RATE_EDIT_OP
									ELSE S.RATE_EDIT
								END RATE_EDIT,
								IS_APPOINTMENT,
								COALESCE(
									NULLIF(
										CASE
											WHEN 1 = 5 THEN CS2.CONC_PCT
											ELSE SP.DISCOUNT_PERCENT
										END,
										0
									),
									SP.DISCOUNT_PERCENT
								) AS CMP_DISC_PCNT,

								COALESCE(CS.EMP_PER, EMP_PERCENT) EMP_PERCENT,

									(
										COALESCE(CS.PRICE, SP.PRICE) * COALESCE(CS.EMP_PER, CC.EMP_PERCENT)

									) / 100
