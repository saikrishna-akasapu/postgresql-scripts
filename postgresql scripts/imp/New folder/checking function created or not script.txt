WITH
	INPUT_funcs AS (
		SELECT
			UNNEST(
				ARRAY[
 'fun_get_lb_dispatch_time',
 'fun_get_lb_dispatch_time_back',
 'fun_get_lb_dispatch_time_his_lab',
 'func_charge_increment',
 'func_get_regtoken_no',
 'get_bill_detid_bybillsrvid',
 'get_format_where_to_json',
 'getalphavalue',
 'getdays',
 'getlocaltime',
 'getmonthname',
  'getmonthnumber',
 'getopeningbalance',
 'gettrailbalancenew',
 'hl7_outbound_profile_name',
 'initcap',
 'least'
 
				]
			) AS func_NAME
	),
	matched_funcs as (
         select p.proname,
		 n.nspname
		 from pg_proc p
		 join pg_namespace n on n.oid=p.pronamespace
		 where n.nspname ='stpl_hims_p_m_dbo' or  n.nspname ='stpl_hims_p_m_acc' and p.prokind='f'
	),
	results as (

SELECT
	I.func_NAME,
	CASE
		WHEN m.PRONAME IS NOT NULL THEN 'created'
		ELSE 'not created'
	END AS STATUS
FROM
	INPUT_funcs I
	LEFT JOIN matched_funcs m ON i.func_name = m.proname
	)

	select * from results
	union all
	select 'Total exists' as func_name,
	count(*)::text as status
	from results where status='created'
	union all
	select 'not created',
	count(*)::text 
	from results where status='not created';
